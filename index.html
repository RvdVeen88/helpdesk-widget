const https = require('https'); module.exports = async function (context, req) { const secret = process.env.AFAS_SECRET_KEY; const copilotEndpoint = process.env.COPILOT_TOKEN_ENDPOINT; // NIEUW: Lees de data nu uit de POST-body in plaats van de URL-balk const tokenUrl = (req.body && req.body.tokenurl) || req.query.tokenurl; const code = (req.body && req.body.code) || req.query.code; if (!secret || !tokenUrl || !code) { context.res = { status: 400, headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" }, body: { error: "Ontbrekende parameters (secret, tokenUrl of code mist in de body)." } }; return; } try { // 1. Wissel code in bij AFAS const afasData = await new Promise((resolve, reject) => { const data = `secret=${encodeURIComponent(secret)}&code=${encodeURIComponent(code)}`; const options = { method: 'POST', headers: { 'Content-Type': 'application/x-www-form-urlencoded', 'Content-Length': Buffer.byteLength(data) } }; const r = https.request(tokenUrl, options, (res) => { let body = ''; res.on('data', (c) => { body += c; }); res.on('end', () => { if (res.statusCode === 200) { try { resolve(JSON.parse(body)); } catch (e) { reject(new Error("AFAS gaf ongeldige JSON terug: " + body)); } } else { reject(new Error("AFAS weigerde de aanvraag (status " + res.statusCode + "): " + body)); } }); }); r.on('error', reject); r.write(data); r.end(); }); // 2. Haal Direct Line Token op bij Copilot const copilotResponse = await new Promise((resolve, reject) => { if (!copilotEndpoint) { reject(new Error("COPILOT_TOKEN_ENDPOINT is niet ingesteld.")); return; } https.get(copilotEndpoint, (res) => { let data = ''; res.on('data', (c) => { data += c; }); res.on('end', () => { if (res.statusCode === 200) { try { resolve(JSON.parse(data)); } catch (e) { reject(new Error("Copilot gaf ongeldige JSON terug.")); } } else { reject(new Error("Copilot weigerde de aanvraag: " + res.statusCode)); } }); }).on('error', reject).end(); }); // 3. Stuur succesvol terug context.res = { status: 200, headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" }, body: copilotResponse }; } catch (error) { context.log("Fout opgetreden:", error); context.res = { status: 500, headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" }, body: { message: "De Azure Function is gecrasht.", foutmelding: error.message, stacktrace: error.stack } }; } };
